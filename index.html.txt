    function registerSubscriber() {
      const role = document.querySelector('input[name="subRole"]:checked').value;
      const fullName = document.getElementById('regFullName').value.trim();
      const address = document.getElementById('regAddress').value.trim();
      const phone = document.getElementById('regPhone').value.trim();
      const password = document.getElementById('regPassword').value.trim();
      const shopName = document.getElementById('regShopName').value.trim();
      const verifyPhone = document.getElementById('regShopVerifyPhone').value.trim();

      if (!fullName || !address || !phone || !password) {
        return alert('يرجى ملء الاسم والعنوان ورقم الهاتف وكلمة السر.');
      }
      if (role === 'shop' && (!shopName || !verifyPhone)) {
        return alert('يجب إدخال اسم المحل ورقم للتأكيد لصاحب المحل.');
      }

      // الجميع بانتظار موافقة المدير حصرياً
      const newSub = {
        id: Date.now(),
        role,
        fullName,
        shopName: role === 'shop' ? shopName : '-',
        verifyPhone: role === 'shop' ? verifyPhone : '-',
        address,
        phone,
        password,
        status: 'pending' // معلق بانتظار المدير
      };

      subscribers.push(newSub);
      saveDataToStorage();
      renderSubscribersAdminTable();

      alert('تم إرسال طلب اشتراكك بنجاح! حسابك الآن بانتظار الموافقة والتأكيد من مدير النظام ليتم تفعيله.');
      closeModal('subscriberModal');
    }

    function renderSubscribersAdminTable() {
      const tbody = document.getElementById('subscribersTableBody');
      const badge = document.getElementById('pendingBadge');

          : '<span class="bg-blue-100 text-blue-800 px-2 py-0.5 rounded font-bold">فني مركّب</span>';

          ? '<span class="bg-emerald-600 text-white px-2 py-0.5 rounded text-[10px] font-bold">مقبول ومفعّل</span>'
          : '<span class="bg-amber-500 text-slate-900 px-2 py-0.5 rounded text-[10px] font-bold">بانتظار قبول المدير</span>';

        const actionBtn = s.status === 'pending'
          ? `<button onclick="approveSubscriber(${s.id})" class="bg-emerald-600 hover:bg-emerald-700 text-white px-3 py-1 rounded font-bold shadow">✔ قبول وتفعيل</button>`
          : `<button onclick="deleteSubscriber(${s.id})" class="bg-red-100 text-red-700 px-2 py-1 rounded font-bold">حذف</button>`;

        tbody.innerHTML += `
          <tr class="border-b">
            <td class="p-2 border">${roleBadge}</td>
            <td class="p-2 border font-bold">${s.fullName}</td>
            <td class="p-2 border">${s.shopName !== '-' ? `${s.shopName}<br><span class="text-slate-500">تأكيد: ${s.verifyPhone}</span>` : '-'}</td>
            <td class="p-2 border">${s.address}</td>

            <td class="p-2 border">${statusBadge}</td>
            <td class="p-2 border text-center">${actionBtn}</td>
          </tr>
        `;
      });

      if (pendingCount > 0) {
        badge.innerText = pendingCount;
        menuCount.innerText = pendingCount;
        badge.classList.remove('hidden');
        menuCount.classList.remove('hidden');
      } else {
        badge.classList.add('hidden');
        menuCount.classList.add('hidden');
      }
    }

    function renderPriceHistoryPeriods() {
      const prodId = parseInt(document.getElementById('historyProdSelect').value);
      const queryDate = document.getElementById('historyDateQuery').value;
      const p = products.find(x => x.id === prodId);
      if (!p) return;

      const tbody = document.getElementById('historyPeriodsBody');
      tbody.innerHTML = '';
      let matchedPeriod = p.pricePeriods[p.pricePeriods.length - 1];

      p.pricePeriods.forEach((period, index) => {
        const isCurrent = index === p.pricePeriods.length - 1;
        if (queryDate && queryDate >= period.from && (isCurrent || queryDate <= period.to)) {
          matchedPeriod = period;
        }
        tbody.innerHTML += `
          <tr class="border-b ${isCurrent ? 'bg-emerald-50 font-bold' : ''}">
            <td class="p-2 border">من <span class="text-blue-700">${period.from}</span> إلى <span class="text-emerald-700">${period.to}</span></td>
            <td class="p-2 border">${period.retail} $</td>
            <td class="p-2 border text-blue-700">${period.installer} $</td>
            <td class="p-2 border text-emerald-700">${period.shop} $</td>
            <td class="p-2 border">${isCurrent ? 'السعر الحالي' : 'سابق'}</td>
          </tr>
        `;
      });

      document.getElementById('historyQueryResult').innerHTML = `
        السعر في تاريخ (${queryDate || 'اليوم'}): المفرق: ${matchedPeriod.retail}$ | المركّب: ${matchedPeriod.installer}$ | المحل: ${matchedPeriod.shop}$
      `;
    }

    function loadProductForEdit() {
      const prodId = parseInt(document.getElementById('editProdSelect').value);

      document.getElementById('editRetailPrice').value = p.prices.retail;

    function updateProductPriceWithPeriod() {
      const prodId = parseInt(document.getElementById('editProdSelect').value);
      const p = products.find(x => x.id === prodId);
      const r = parseFloat(document.getElementById('editRetailPrice').value);
      const ins = parseFloat(document.getElementById('editInstallerPrice').value);
      const shp = parseFloat(document.getElementById('editShopPrice').value);
      const newDate = document.getElementById('editEffectiveDate').value;

      if (!newDate || isNaN(r) || isNaN(ins) || isNaN(shp)) return alert('يرجى تعبئة الأسعار والتاريخ.');

      if (p.pricePeriods.length > 0) {
        p.pricePeriods[p.pricePeriods.length - 1].to = newDate;
      }
      p.pricePeriods.push({ from: newDate, to: 'الآن (السعر الحالي)', retail: r, installer: ins, shop: shp });
      p.prices = { retail: r, installer: ins, shop: shp };

      saveDataToStorage();
      renderProducts();
      renderPriceHistoryPeriods();
      alert('تم تحديث السعر وتوثيق الفترة التاريخية!');
    }

    function addNewProduct() {
      const name = document.getElementById('newProdName').value.trim();
      const cat = document.getElementById('newProdCat').value;
      const desc = document.getElementById('newProdDesc').value.trim();
      const r = parseFloat(document.getElementById('newProdRetail').value);
      const ins = parseFloat(document.getElementById('newProdInstaller').value);
      const shp = parseFloat(document.getElementById('newProdShop').value);
      const textLinks = document.getElementById('newProdImgLinks').value.trim();

      if (!name || isNaN(r) || isNaN(ins) || isNaN(shp)) return alert('يرجى تعبئة اسم المنتج والأسعار.');

      const catNames = { solar: 'طاقة شمسية', cctv: 'كاميرات مراقبة', alarm: 'إنذار مبكر' };
      const today = new Date().toISOString().split('T')[0];
      const imgs = textLinks ? textLinks.split(',').map(s => s.trim()).filter(Boolean) : ['https://images.unsplash.com/photo-1509391366360-2e959784a276?w=600'];

      products.push({
        id: Date.now(),
        category: cat,
        catName: catNames[cat],
        name,
        desc: desc || 'منتج مكفول.',
        prices: { retail: r, installer: ins, shop: shp },
        images: imgs,
        pricePeriods: [{ from: today, to: 'الآن (السعر الحالي)', retail: r, installer: ins, shop: shp }]
      });

      saveDataToStorage();
      initSelectBoxes();
      renderProducts();
      alert('تمت إضافة المنتج بنجاح!');
    }

    function addToCart(id) {
      const existing = cart.find(i => i.id === id);
      if (existing) existing.qty += 1;
      else cart.push({ id, qty: 1 });
      updateCartUI();
    }

    function updateCartUI() {
      let total = 0, count = 0;
      const list = document.getElementById('cartItemsList');
      list.innerHTML = '';

      cart.forEach((item, idx) => {
        const p = products.find(x => x.id === item.id);
        if (!p) return;
        const price = getActivePrice(p);
        const sub = price * item.qty;
        total += sub;
        count += item.qty;

        list.innerHTML += `
          <div class="flex justify-between items-center bg-slate-50 p-2 rounded-lg border text-xs">
            <div>
              <div class="font-bold text-slate-800">${p.name}</div>
              <div class="text-slate-500">${price} $ × ${item.qty} = <strong class="text-emerald-600">${sub} $</strong></div>
            </div>
            <div class="flex items-center gap-2">
              <button onclick="changeQty(${idx}, -1)" class="bg-slate-200 w-6 h-6 rounded font-bold">-</button>
              <span class="font-bold">${item.qty}</span>
              <button onclick="changeQty(${idx}, 1)" class="bg-slate-200 w-6 h-6 rounded font-bold">+</button>
            </div>
          </div>
        `;
      });

      document.getElementById('cartCount').innerText = count;
      document.getElementById('bottomTotal').innerText = total + ' $';
      document.getElementById('cartTotalAmount').innerText = total + ' $';
    }

    function changeQty(idx, delta) {
      cart[idx].qty += delta;
      if (cart[idx].qty <= 0) cart.splice(idx, 1);
      updateCartUI();
    }

    function togglePaymentBox() {
      const method = document.getElementById('paymentMethod').value;
      const box = document.getElementById('payInstructionBox');
      const txt = document.getElementById('payAccountText');
      if (method.includes('شام كاش')) {
        box.classList.remove('hidden');
        txt.innerText = 'حساب شام كاش: ' + storeSettings.shamCash;
      } else if (method.includes('مصاري')) {
        box.classList.remove('hidden');
        txt.innerText = 'محفظة مصاري: ' + storeSettings.masari;
      } else if (method.includes('حوالة')) {
        box.classList.remove('hidden');
        txt.innerText = 'بيانات التحويل: ' + storeSettings.otherPay;
      } else {
        box.classList.add('hidden');
      }
    }

    function sendWhatsAppOrder() {
      if (cart.length === 0) return alert('السلة فارغة!');
      const name = document.getElementById('orderCustName').value || 'غير محدد';
      const address = document.getElementById('orderCustAddress').value || 'غير محدد';
      const phone = document.getElementById('orderCustPhone').value || 'غير محدد';
      const payMethod = document.getElementById('paymentMethod').value;
      const ref = document.getElementById('transferNumber').value;

      const userType = currentSubscriber && currentSubscriber.status === 'approved'
        ? (currentSubscriber.role === 'installer' ? 'مشترك: فني مركّب' : `مشترك: صاحب محل (${currentSubscriber.shopName})`)
        : 'زبون مفرق';

      let msg = `*طلب جديد - ${storeSettings.name}*\n`;
      msg += `--------------------------------\n`;
      msg += `*الفئة:* ${userType}\n`;
      msg += `*الاسم:* ${name}\n`;
      msg += `*العنوان:* ${address}\n`;
      msg += `*الهاتف:* ${phone}\n`;
      msg += `*طريقة الدفع:* ${payMethod}\n`;
      if (ref) msg += `*رقم إشعار التحويل:* ${ref}\n`;
      msg += `--------------------------------\n*الطلبيات:*\n`;

      let total = 0;
      cart.forEach(item => {
        const p = products.find(x => x.id === item.id);
        const price = getActivePrice(p);
        const sub = price * item.qty;
        total += sub;
        msg += `• ${p.name} (العدد: ${item.qty}) = ${sub} $\n`;
      });

      msg += `--------------------------------\n*المجموع:* ${total} $\n`;
      window.open(`https://wa.me/${storeSettings.phone}?text=${encodeURIComponent(msg)}`, '_blank');
    }
